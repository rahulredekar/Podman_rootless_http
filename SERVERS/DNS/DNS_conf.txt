# 🔵 STEP 2 — PRIMARY DNS (MASTER)

## 📍 RUN ON: pri1 (192.168.83.4)

---

```bash id="p1"
sudo dnf install bind bind-utils -y
```

---

```bash id="p2"
sudo semanage port -a -t dns_port_t -p tcp 53 2>/dev/null || sudo semanage port -m -t dns_port_t -p tcp 53
sudo semanage port -a -t dns_port_t -p udp 53 2>/dev/null || sudo semanage port -m -t dns_port_t -p udp 53
```

---

```bash id="p3"
sudo semanage fcontext -a -t named_zone_t "/var/named(/.*)?"
sudo restorecon -Rv /var/named
```

---

```bash id="p4"
sudo vim /etc/named.conf
```

---

```conf id="p5"
options {
    directory "/var/named";
    listen-on port 53 { any; };

    allow-query { any; };
    recursion no;

    allow-transfer { 192.168.83.5; };

    dnssec-validation auto;
};

zone "vaibhav.com" IN {
    type master;
    file "vaibhav.com.zone.signed";
};
```

---

## DNSSEC (PRI1 ONLY)

```bash id="p6"
cd /var/named
sudo dnssec-keygen -a RSASHA256 -b 2048 -n ZONE vaibhav.com
sudo dnssec-keygen -f KSK -a RSASHA256 -b 4096 -n ZONE vaibhav.com
```

---

```bash id="p7"
sudo dnssec-signzone -A -o vaibhav.com -f /var/named/vaibhav.com.zone.signed /var/named/vaibhav.com.zone
```

---

```bash id="p8"
sudo rndc reload
```

---

```bash id="p9"
sudo systemctl enable --now named
sudo systemctl restart named
```

---

```bash id="p10"
sudo firewall-cmd --permanent --add-service=dns
sudo firewall-cmd --reload
```
